## 코드1

```python
from collections import deque

n = list(map(int, input().split()))
queue = deque(n)
a = []
for i in range(4) :
    queue.rotate(-1)
    a.append(int("".join(map(str, queue))))
b = set()
count = 0
for i in range(1111,min(a)) :
    queue = deque(str(i))
    isTrue = True
    for k in range(4) :
        queue.rotate(-1)
        t = int("".join(map(str, queue)))
        if t in b or '0' in queue :
            isTrue = False
    if isTrue :
        b.add(i)
        count += 1
print(count + 1)
```

n = list(map(int, input().split()))
a = []
for i in range(4) :
    n.append(n[0])
    n.remove(n[0])
    a.append(int("".join(map(str, n))))
b = []
for i in range(1111, min(a)) :
    l = list(map(str, str(i)))
    isTrue = True
    if '0' in l :
        continue
    for k in range(4) :
        l.append(l[0])
        l.remove(l[0])
        tt = int("".join(map(str, l)))
        if tt in b :
            isTrue = False
            break
    if isTrue :
        b.append(i)
print(len(b)+1)
