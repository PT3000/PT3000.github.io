## 구현

- 퀸,나이트,폰의 위치를 1,2,3으로 표시하여 2차원 배열에 저장하였다.
- 여기서 관건은 퀸의 움직임인데 퀸의 위치를 제외한 가로,세로 대각선의 허용 범위를 1칸씩 검사한다.
- 검사할때마다. 검사한 곳은 9로 표기하고, 다른 말들이 검사했을 때, 검사한 곳이라도 다시 재검사를 한다. → 다른 말들이 지나간 곳이라 해도, 다른 말들도 지나갈 수 있기 떄문이다.
- 나이트는 좌표값을 저장하여 검사하였다.

## 코드

```cpp
#include <iostream>
#include <vector>
using namespace std;

int main() {
    int n, m;
    int chess;
    int x, y;
    int dx = 0;
    int dy = 0;
    int correct = 0;
    bool brk = false;
    int right = 0;
    int left = 0;
    cin >> n >> m;
    // 퀸=1 || 나이트=2 || 폰=3 || 검사한 위치=9
    vector<vector<int>> chessBoard(n, vector<int>(m, 0));
    vector<pair<int, int> > knightXY = {
        {-1, -2}, {-2, -1},
        {-2, 1}, {-1, 2},
        {1, -2}, {2, -1},
        {2, 1}, {1, 2}
    };


    for (int i = 0; i < 3; i++) {
        cin >> chess;
        for (int k = 0; k < chess; k++) {
            cin >> x >> y;
            chessBoard[x - 1][y - 1] = i + 1; //각 말들의 좌표를 보드에 저장
        }
    }

    for (int i = 0; i < n; i++) {
        for (int k = 0; k < m; k++) {
            if (chessBoard[i][k] == 1) {
                //퀸일때 : 가로,세로,대각선에 장매물이 없어야함.
                for (int x1 = k - 1; x1 > -1; x1--) {
                    //가로(행) 왼쪽 검사 -> 행을 검사할때는 열의 위치가 움직여야 한다. 열: k
                    if (chessBoard[i][x1] != 0 && chessBoard[i][x1] != 9) {
                        break;
                    } else {
                        chessBoard[i][x1] = 9;
                    }
                }
                for (int x1 = k + 1; x1 < m; x1++) {
                    //가로(행) 오른쪽 검사
                    if (chessBoard[i][x1] != 0 && chessBoard[i][x1] != 9) {
                        break;
                    } else {
                        chessBoard[i][x1] = 9;
                    }
                }
                for (int y1 = i - 1; y1 > -1; y1--) {
                    //세로(열) 위쪽 검사 -> 열을 검사할때는 행의 위치가 움직여야 한다. 행 : i
                    if (chessBoard[y1][k] != 0 && chessBoard[y1][k] != 9) {
                        break;
                    } else {
                        chessBoard[y1][k] = 9;
                    }
                }
                for (int y1 = i + 1; y1 < n; y1++) {
                    //세로(열) 아래쪽 검사
                    if (chessBoard[y1][k] != 0 && chessBoard[y1][k] != 9) {
                        break;
                    } else {
                        chessBoard[y1][k] = 9;
                    }
                }
                left = k - 1;
                for (int up = i - 1; up > -1; up--) {
                    // 대각선 검사 -> 행을 기준으로 삼아야 한다.
                    if (left < 0) {
                        break;
                    }
                    if (chessBoard[up][left] != 0 && chessBoard[up][left] != 9) {
                        break;
                    } else {
                        chessBoard[up][left] = 9;
                    }
                    left--;
                }
                right = k + 1;
                for (int up = i - 1; up > -1; up--) {
                    if (right < 0) {
                        break;
                    }
                    if (chessBoard[up][right] != 0 && chessBoard[up][right] != 9) {
                        break;
                    } else {
                        chessBoard[up][right] = 9;
                    }
                    right++;
                }
                left = k - 1;
                for (int down = i + 1; down < n; down++) {
                    if (left < 0) {
                        break;
                    }
                    //<- 이 부분 이상함
                    if (chessBoard[down][left] != 0 && chessBoard[down][left] != 9) {
                        break;
                    } else {
                        chessBoard[down][left] = 9;
                    }
                    left--;
                }
                right = k + 1;
                for (int down = i + 1; down < n; down++) {
                    if (right < 0) {
                        break;
                    }
                    if (chessBoard[down][right] != 0 && chessBoard[down][right] != 9) {
                        break;
                    } else {
                        chessBoard[down][right] = 9;
                    }
                    right++;
                }
            }
            if (chessBoard[i][k] == 2) {
                //나이트일때
                dy = i;
                dx = k;
                for (const auto &a: knightXY) {
                    dy = i + a.first;
                    dx = k + a.second;
                    if (0 <= dy && dy < n && 0 <= dx && dx < m) {
                        if (chessBoard[dy][dx] == 0) {
                            chessBoard[dy][dx] = 9;
                        }
                    }
                }
            }
        }
    }
    for (int i = 0; i < n; i++) {
        for (int k = 0; k < m; k++) {
            if (chessBoard[i][k] == 0) {
                correct++;
            }
        }
    }
    cout << correct << endl;
    return 0;
}
```
