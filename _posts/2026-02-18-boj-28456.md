```python
import copy

n = int(input())
a = []
b = []
d = [[0 for _ in range(n)] for _ in range(n)]
for i in range(n) :
    n2 = list(map(int, input().split()))
    a.append(n2)

q = int(input())

i = 0
j = 0

for t in range(q) :
    c = list(map(int, input().split()))
    if len(c) > 1 :
        save = a[c[1]-1][-1]
        a[c[1]-1].pop()

        a[c[1]-1].insert(0, save)
    else :
        for i in range(1,n+1) :
            for j in range(1,n+1) :
                d[j-1][n-i] = a[i-1][j-1]
        a = copy.deepcopy(d)

for row in a:
    print(" ".join(map(str, row)))
```
