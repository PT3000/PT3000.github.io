```python
n = int(input())
n2 = list(map(int ,input().split()))  ## a ~ f까지 입력받기
l = []
def solution() :
    if n == 1 :
        n2.remove(max(n2))
        correct = sum(n2)
        return correct

    l.append(min(n2[0],n2[5]))
    l.append(min(n2[1],n2[4]))
    l.append(min(n2[2],n2[3]))
    l.sort()

    a = l[0]
    b = l[1]
    c = l[2]

    correct = 4 * (a * n * (n-1) + n * b) + 4 * c + (n-2) * (n-2) * a + 4 * (n-2) * b
    return correct

print(solution())
```
