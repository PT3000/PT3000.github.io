```python
n,m = map(int, input().split())
correct = []
correct2 = []
def backtracking() :
    if len(correct) == m :
        if sorted(correct) not in correct2 :
            print(*correct, end='\n')
            correct2.append(sorted(correct))
        return
    for i in range(1,n+1) :
        if i not in correct :
            correct.append(i)
            backtracking()
            correct.pop()

backtracking()
```
