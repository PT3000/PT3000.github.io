```python
from collections import defaultdict

num = int(input())
correct = [list(map(int, input().split())) for _ in range(num)]
student = []
stu_list = []
stu_list = defaultdict(list) ## 딕녀너리의 일종인, 디폴트딕트를 사용

for i in range(0,num) :
    student.append(0)

def play(i,t) :
    for k in range(i+1,num) :
        if correct[i][t] == correct[k][t] :
            if k not in stu_list[i] :
                student[i] += 1
                student[k] += 1
                stu_list[i].append(k)
                stu_list[k].append(i)

for t in range(0,5) :
    for i in range(0,num) :
        play(i,t)

print(student.index(max(student))+1)
```
